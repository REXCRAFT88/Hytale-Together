cmake_minimum_required(VERSION 3.20)
project(HytaleInput LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(deps/minhook)
include_directories(deps/SDL/include src)
add_library(SDL3 SHARED IMPORTED)
set_target_properties(SDL3 PROPERTIES IMPORTED_IMPLIB "${CMAKE_SOURCE_DIR}/SDL3.lib" IMPORTED_LOCATION "${CMAKE_SOURCE_DIR}/SDL3.dll")
add_library(HytaleInput SHARED src/adapter.cpp)
set_target_properties(HytaleInput PROPERTIES OUTPUT_NAME "HytaleInput-v5")
target_link_libraries(HytaleInput PRIVATE minhook SDL3 user32 gdi32 ws2_32 iphlpapi)
add_executable(HytaleInputHost src/host.cpp src/app.rc)
target_link_libraries(HytaleInputHost PRIVATE SDL3 user32 advapi32 ws2_32)
add_executable(HytaleInputFixture src/fixture.cpp)
target_link_libraries(HytaleInputFixture PRIVATE SDL3 user32 ws2_32)
add_library(HytaleLaunch SHARED src/launch_adapter.cpp)
target_link_libraries(HytaleLaunch PRIVATE minhook shell32)
add_executable(HytaleLaunchHost src/launch_host.cpp src/app.rc)
add_executable(HytaleLaunchFixture src/launch_fixture.cpp)
target_link_libraries(HytaleLaunchFixture PRIVATE minhook shell32)
foreach(t HytaleInput HytaleInputHost HytaleInputFixture HytaleLaunch HytaleLaunchHost HytaleLaunchFixture)
  target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_options(${t} PRIVATE "/DYNAMICBASE" "/NXCOMPAT" "/HIGHENTROPYVA")
endforeach()
